KEY CONFIGURATION PATTERNS

Pattern 1: Local Model (Ollama)
  config = {
      "llm": {"model": "ollama/llama3.2", "model_tokens": 4096},
      "headless": True, "verbose": True
  }

Pattern 2: API Model with Rate Limiting
  config = {
      "llm": {
          "api_key": "sk-...",
          "model": "openai/gpt-4o-mini",
          "rate_limit": {"requests_per_second": 5, "max_retries": 3}
      }
  }

Pattern 3: Advanced with Quality Control
  config = {
      "llm": {...},
      "reattempt": True,          # Retry on failure
      "reasoning": True,          # Multi-step thinking
      "html_mode": False,         # Use Markdown
      "additional_info": "...",   # Custom system prompt
      "storage_state": "auth.json" # Cookie/session persistence
  }
